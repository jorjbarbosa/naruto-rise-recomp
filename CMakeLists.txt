# Code shared by the generated sources and the runtime: CPU baseline and the hooks in ppc_detail.h.
# Each module adds its own generated directory, which holds its ppc_config.h.
add_library(ppc_common INTERFACE)
target_include_directories(ppc_common INTERFACE
    "${CMAKE_SOURCE_DIR}/src/ppc"
    "${XENON_RECOMP_DIR}/thirdparty/simde")
target_compile_definitions(ppc_common INTERFACE PPC_INCLUDE_DETAIL NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_compile_options(ppc_common INTERFACE /arch:AVX)

# add_ppc_module(<name> <generated dir> <guest file name> <mapping table>): an object library with the
# recompiled code of one guest module, plus its PPCModuleInfo (g_ppcModule_<name>, see ppc_module.h),
# compiled against that module's ppc_config.h. Line tables only, so crash reports can name the guest
# function.
set(PPC_MODULE_OBJECTS "")
set(PPC_SHARED_HEADERS "")
function(add_ppc_module name dir file mappings)
    file(GLOB sources CONFIGURE_DEPENDS "${dir}/*.cpp")
    if(NOT sources)
        message(FATAL_ERROR "No generated code in ${dir}; run scripts/recompile.ps1 first")
    endif()
    set(info "${CMAKE_BINARY_DIR}/generated/ppc_module_${name}.cpp")
    file(CONFIGURE OUTPUT "${info}" CONTENT [=[
// Generated by CMake (add_ppc_module): describes the recompiled module @name@.
#include "ppc_recomp_shared.h"
#include "ppc_module.h"

extern PPCFuncMapping @mappings@[];

extern const PPCModuleInfo g_ppcModule_@name@ = {
    "@file@", PPC_IMAGE_BASE, PPC_IMAGE_SIZE, PPC_CODE_BASE, PPC_CODE_SIZE, @mappings@ };
]=] @ONLY)
    add_library(ppc_${name} OBJECT ${sources} "${info}")
    target_include_directories(ppc_${name} PRIVATE "${dir}")
    target_link_libraries(ppc_${name} PRIVATE ppc_common)
    target_compile_options(ppc_${name} PRIVATE -Wno-everything -gline-tables-only)
    set(PPC_MODULE_OBJECTS ${PPC_MODULE_OBJECTS} $<TARGET_OBJECTS:ppc_${name}> PARENT_SCOPE)
    set(PPC_SHARED_HEADERS ${PPC_SHARED_HEADERS} "${dir}/ppc_recomp_shared.h" PARENT_SCOPE)
endfunction()
add_ppc_module(default "${PPC_OUTPUT_DIR}" "default.xex" PPCFuncMappings)
add_ppc_module(ai2c "${PPC_AI2C_OUTPUT_DIR}" "AI2C.dll" PPCFuncMappings_ai2c)

# FFmpeg's XMA1 decoder (libavutil + libavcodec with the xma1 decoder only), built from the sources
# and configuration that scripts/ffmpeg-configure.ps1 wrote to cmake/ffmpeg, as FFmpeg's make would.
set(FFMPEG_DIR "${CMAKE_SOURCE_DIR}/thirdparty/ffmpeg")
set(FFMPEG_CONFIG_DIR "${CMAKE_SOURCE_DIR}/cmake/ffmpeg")
include("${FFMPEG_CONFIG_DIR}/sources.cmake")
function(add_ffmpeg_library name)
    list(TRANSFORM ARGN PREPEND "${FFMPEG_DIR}/")
    add_library(ffmpeg_${name} STATIC ${ARGN})
    target_include_directories(ffmpeg_${name} PUBLIC "${FFMPEG_CONFIG_DIR}" "${FFMPEG_DIR}"
        PRIVATE "${FFMPEG_DIR}/compat/stdbit")
    target_compile_definitions(ffmpeg_${name} PRIVATE HAVE_AV_CONFIG_H BUILDING_${name} _ISOC11_SOURCE
        _FILE_OFFSET_BITS=64 _LARGEFILE_SOURCE WIN32_LEAN_AND_MEAN _USE_MATH_DEFINES _CRT_SECURE_NO_WARNINGS
        _CRT_NONSTDC_NO_WARNINGS _WIN32_WINNT=0x0600)
    target_compile_options(ffmpeg_${name} PRIVATE /std:c17 -utf-8 -Wno-everything)
endfunction()
add_ffmpeg_library(avutil ${FFMPEG_AVUTIL_SOURCES})
target_link_libraries(ffmpeg_avutil PUBLIC user32 bcrypt)
add_ffmpeg_library(avcodec ${FFMPEG_AVCODEC_SOURCES})
# codec_list.c and the other lists are included by allcodecs.c and its siblings, from the config dir.
target_link_libraries(ffmpeg_avcodec PUBLIC ffmpeg_avutil ole32 user32)

# Every kernel/XAM import of the recompiled modules gets a weak logging stub (src/kernel/stubs.cpp);
# the runtime overrides the implemented ones. The list comes from the generated headers.
set(IMPORT_NAMES "")
foreach(header IN LISTS PPC_SHARED_HEADERS)
    file(STRINGS "${header}" PPC_EXTERN_LINES REGEX "^PPC_EXTERN_FUNC\\(__imp__")
    foreach(line IN LISTS PPC_EXTERN_LINES)
        string(REGEX REPLACE "^PPC_EXTERN_FUNC\\(__imp__(.*)\\);$" "\\1" name "${line}")
        list(APPEND IMPORT_NAMES "${name}")
    endforeach()
endforeach()
list(REMOVE_DUPLICATES IMPORT_NAMES)
set(IMPORTS_INC "")
foreach(name IN LISTS IMPORT_NAMES)
    string(APPEND IMPORTS_INC "XBOX_IMPORT(${name})\n")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/generated/imports.inc" CONTENT "${IMPORTS_INC}")

set(DXC "${CMAKE_SOURCE_DIR}/tools/XenosRecomp/thirdparty/dxc-bin/bin/x64/dxc.exe")

file(MAKE_DIRECTORY "${GPU_SHADER_OUTPUT}")
set(GPU_SHADER_HEADERS "")
# add_gpu_shader(<header name> <array name> <source> <profile> [dxc options...])
function(add_gpu_shader name variable source profile)
    set(output "${GPU_SHADER_OUTPUT}/${name}.h")
    add_custom_command(OUTPUT "${output}"
        COMMAND "${DXC}" -nologo -T ${profile} -E main -O3 -Vn ${variable} -Fh "${output}" ${ARGN} "${GPU_SHADER_DIR}/${source}"
        DEPENDS "${GPU_SHADER_DIR}/${source}"
        COMMENT "Compiling shader ${name}"
        VERBATIM)
    set(GPU_SHADER_HEADERS ${GPU_SHADER_HEADERS} "${output}" PARENT_SCOPE)
endfunction()
add_gpu_shader(fullscreen_vs kFullscreenVs fullscreen_vs.hlsl vs_6_0)
add_gpu_shader(present_ps kPresentPs present_ps.hlsl ps_6_0)
add_gpu_shader(resolve_color_ms_ps kResolveColorMsPs resolve_ps.hlsl ps_6_0 -DMULTISAMPLE=1)
add_gpu_shader(resolve_depth_ps kResolveDepthPs resolve_ps.hlsl ps_6_0 -DDEPTH=1)
add_gpu_shader(resolve_depth_ms_ps kResolveDepthMsPs resolve_ps.hlsl ps_6_0 -DDEPTH=1 -DMULTISAMPLE=1)
