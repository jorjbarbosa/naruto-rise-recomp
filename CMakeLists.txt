# Code shared by the generated sources and the runtime: CPU baseline and the hooks in ppc_detail.h.
# Each module adds its own generated directory, which holds its ppc_config.h.
add_library(ppc_common INTERFACE)
target_include_directories(ppc_common INTERFACE
    "${CMAKE_SOURCE_DIR}/src/ppc"
    "${XENON_RECOMP_DIR}/thirdparty/simde")
target_compile_definitions(ppc_common INTERFACE PPC_INCLUDE_DETAIL NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
# The same CPU baseline in the spelling of each driver: clang-cl on Windows, clang elsewhere. clang-cl
# turns /arch:AVX into the Sandy Bridge feature set with generic tuning. It also does not assume
# strict aliasing, and that is what the generated code and the accessors of guest memory have always
# been compiled and tested with, so clang is told the same.
if(MSVC)
    target_compile_options(ppc_common INTERFACE /arch:AVX)
else()
    target_compile_options(ppc_common INTERFACE -march=sandybridge -mtune=generic -fno-strict-aliasing)
endif()

# add_ppc_module(<name> <generated dir> <guest file name> <mapping table>): an object library with the
# recompiled code of one guest module, plus its PPCModuleInfo (g_ppcModule_<name>, see ppc_module.h),
# compiled against that module's ppc_config.h. Line tables only, so crash reports can name the guest
# function.
set(PPC_MODULE_OBJECTS "")
set(PPC_SHARED_HEADERS "")
function(add_ppc_module name dir file mappings)
    file(GLOB sources CONFIGURE_DEPENDS "${dir}/*.cpp")
    if(NOT sources)
        message(FATAL_ERROR "No generated code in ${dir}; run scripts/recompile.ps1 (recompile.sh on Linux) first")
    endif()
    set(info "${CMAKE_BINARY_DIR}/generated/ppc_module_${name}.cpp")
    file(CONFIGURE OUTPUT "${info}" CONTENT [=[
// Generated by CMake (add_ppc_module): describes the recompiled module @name@.
#include "ppc_recomp_shared.h"
#include "ppc_module.h"

extern PPCFuncMapping @mappings@[];

extern const PPCModuleInfo g_ppcModule_@name@ = {
    "@file@", PPC_IMAGE_BASE, PPC_IMAGE_SIZE, PPC_CODE_BASE, PPC_CODE_SIZE, @mappings@ };
]=] @ONLY)
    add_library(ppc_${name} OBJECT ${sources} "${info}")
    target_include_directories(ppc_${name} PRIVATE "${dir}")
    target_link_libraries(ppc_${name} PRIVATE ppc_common)
    target_compile_options(ppc_${name} PRIVATE -Wno-everything -gline-tables-only)
    set(PPC_MODULE_OBJECTS ${PPC_MODULE_OBJECTS} $<TARGET_OBJECTS:ppc_${name}> PARENT_SCOPE)
    set(PPC_SHARED_HEADERS ${PPC_SHARED_HEADERS} "${dir}/ppc_recomp_shared.h" PARENT_SCOPE)
endfunction()
add_ppc_module(default "${PPC_OUTPUT_DIR}" "default.xex" PPCFuncMappings)

# FFmpeg's XMA1 decoder (libavutil + libavcodec with the xma1 decoder only), built from the sources
# and configuration that scripts/ffmpeg-configure.ps1 (Windows) or ffmpeg-configure.sh (Linux) wrote to
# cmake/ffmpeg/<platform>, with the definitions and libraries FFmpeg's configure chose there, as
# FFmpeg's make would.
set(FFMPEG_DIR "${CMAKE_SOURCE_DIR}/thirdparty/ffmpeg")
if(WIN32)
    set(FFMPEG_CONFIG_DIR "${CMAKE_SOURCE_DIR}/cmake/ffmpeg/windows")
else()
    set(FFMPEG_CONFIG_DIR "${CMAKE_SOURCE_DIR}/cmake/ffmpeg/linux")
endif()
include("${FFMPEG_CONFIG_DIR}/sources.cmake")
function(add_ffmpeg_library name)
    list(TRANSFORM ARGN PREPEND "${FFMPEG_DIR}/")
    add_library(ffmpeg_${name} STATIC ${ARGN})
    target_include_directories(ffmpeg_${name} PUBLIC "${FFMPEG_CONFIG_DIR}" "${FFMPEG_DIR}"
        PRIVATE "${FFMPEG_DIR}/compat/stdbit")
    target_compile_definitions(ffmpeg_${name} PRIVATE HAVE_AV_CONFIG_H BUILDING_${name} _ISOC11_SOURCE
        _FILE_OFFSET_BITS=64 _LARGEFILE_SOURCE)
    if(WIN32)
        target_compile_definitions(ffmpeg_${name} PRIVATE WIN32_LEAN_AND_MEAN _USE_MATH_DEFINES
            _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_WARNINGS _WIN32_WINNT=0x0600)
        target_compile_options(ffmpeg_${name} PRIVATE /std:c17 -utf-8 -Wno-everything)
    else()
        target_compile_definitions(ffmpeg_${name} PRIVATE _POSIX_C_SOURCE=200112 _XOPEN_SOURCE=600)
        target_compile_options(ffmpeg_${name} PRIVATE -std=c17 -Wno-everything)
    endif()
endfunction()
add_ffmpeg_library(avutil ${FFMPEG_AVUTIL_SOURCES})
add_ffmpeg_library(avcodec ${FFMPEG_AVCODEC_SOURCES})
# codec_list.c and the other lists are included by allcodecs.c and its siblings, from the config dir.

# Every kernel/XAM import of the recompiled modules gets a weak logging stub (src/kernel/stubs.cpp);
# the runtime overrides the implemented ones. The list comes from the generated headers.
set(IMPORT_NAMES "")
foreach(header IN LISTS PPC_SHARED_HEADERS)
    file(STRINGS "${header}" PPC_EXTERN_LINES REGEX "^PPC_EXTERN_FUNC\\(__imp__")
    foreach(line IN LISTS PPC_EXTERN_LINES)
        string(REGEX REPLACE "^PPC_EXTERN_FUNC\\(__imp__(.*)\\);$" "\\1" name "${line}")
        list(APPEND IMPORT_NAMES "${name}")
    endforeach()
endforeach()
list(REMOVE_DUPLICATES IMPORT_NAMES)
set(IMPORTS_INC "")
foreach(name IN LISTS IMPORT_NAMES)
    string(APPEND IMPORTS_INC "XBOX_IMPORT(${name})\n")
endforeach()
file(CONFIGURE OUTPUT "${CMAKE_BINARY_DIR}/generated/imports.inc" CONTENT "${IMPORTS_INC}")

# The renderer's own passes, compiled with the DXC of the XenosRecomp submodule into a header per
# format: SPIR-V for Vulkan (array name with a Spirv suffix) and, on Windows, DXIL for D3D12 (its
# dxil.dll signs them).
set(DXC_DIR "${CMAKE_SOURCE_DIR}/tools/XenosRecomp/thirdparty/dxc-bin")
if(WIN32)
    set(DXC "${DXC_DIR}/bin/x64/dxc.exe")
else()
    # The Linux DXC finds its libdxcompiler.so beside the others of the submodule only through the
    # loader's path.
    set(DXC "${CMAKE_COMMAND}" -E env "LD_LIBRARY_PATH=${DXC_DIR}/lib/x64" "${DXC_DIR}/bin/x64/dxc-linux")
endif()

set(GPU_SHADER_OUTPUT "${CMAKE_BINARY_DIR}/generated/shaders")
file(MAKE_DIRECTORY "${GPU_SHADER_OUTPUT}")
set(GPU_SHADER_HEADERS "")
# add_gpu_shader(<header name> <array name> <source> <profile> [OPTIONS <dxc options>...]
#     [SPIRV_OPTIONS <dxc options for SPIR-V only>...])
function(add_gpu_shader name variable source profile)
    cmake_parse_arguments(PARSE_ARGV 4 SHADER "" "" "OPTIONS;SPIRV_OPTIONS")
    set(spirv "${GPU_SHADER_OUTPUT}/${name}_spirv.h")
    add_custom_command(OUTPUT "${spirv}"
        COMMAND ${DXC} -nologo -T ${profile} -E main -O3 -spirv -fspv-target-env=vulkan1.2 -Vn ${variable}Spirv
            -Fh "${spirv}" ${SHADER_OPTIONS} ${SHADER_SPIRV_OPTIONS} "${GPU_SHADER_DIR}/${source}"
        DEPENDS "${GPU_SHADER_DIR}/${source}"
        COMMENT "Compiling shader ${name} (SPIR-V)"
        VERBATIM)
    set(headers "${spirv}")
    if(WIN32)
        set(dxil "${GPU_SHADER_OUTPUT}/${name}.h")
        add_custom_command(OUTPUT "${dxil}"
            COMMAND ${DXC} -nologo -T ${profile} -E main -O3 -Vn ${variable} -Fh "${dxil}" ${SHADER_OPTIONS}
                "${GPU_SHADER_DIR}/${source}"
            DEPENDS "${GPU_SHADER_DIR}/${source}"
            COMMENT "Compiling shader ${name} (DXIL)"
            VERBATIM)
        list(APPEND headers "${dxil}")
    endif()
    set(GPU_SHADER_HEADERS ${GPU_SHADER_HEADERS} ${headers} PARENT_SCOPE)
endfunction()
# The translated vertex shaders flip Y for Vulkan's clip space (XenosRecomp passes -fvk-invert-y), and
# so does the passes' one: every viewport keeps a positive height on both APIs.
add_gpu_shader(fullscreen_vs kFullscreenVs fullscreen_vs.hlsl vs_6_0 SPIRV_OPTIONS -fvk-invert-y)
add_gpu_shader(present_ps kPresentPs present_ps.hlsl ps_6_0)
add_gpu_shader(resolve_color_ms_ps kResolveColorMsPs resolve_ps.hlsl ps_6_0 OPTIONS -DMULTISAMPLE=1)
add_gpu_shader(resolve_depth_ps kResolveDepthPs resolve_ps.hlsl ps_6_0 OPTIONS -DDEPTH=1)
add_gpu_shader(resolve_depth_ms_ps kResolveDepthMsPs resolve_ps.hlsl ps_6_0 OPTIONS -DDEPTH=1 -DMULTISAMPLE=1)
